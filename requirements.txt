# =============================================================================
# Procedural 3D Asset Forge — generation dependencies (Python 3.11)
# -----------------------------------------------------------------------------
# Every model in public/assets/models/ is produced by these three libraries
# alone. All of them publish manylinux / macOS / Windows wheels for CPython
# 3.11, so `pip install --prefer-binary` never needs gcc/gfortran.
#
# Blender (`bpy`) and `fast-simplification` are deliberately NOT required:
# geometry is authored analytically with numpy + scipy + trimesh, and polygon
# budget enforcement uses a deterministic vertex-cluster pass instead of a
# quadric simplifier, which keeps local and CI output byte-identical.
# =============================================================================

# Vectorized geometry math (loft profiles, heightfields, normals, vertex colors)
numpy>=1.26.4,<3.0.0

# Cubic spline anatomical profiles & 3D rotation helpers for the character rig
scipy>=1.11.4,<2.0.0

# In-memory mesh assembly, watertight primitives, and GLB round-trip validation
trimesh>=4.0.0,<6.0.0
